cmake_minimum_required(VERSION 3.22)
project(bovary LANGUAGES C CXX)
option(BOVARY_CPU_ONLY "Build inference with only the statically linked CPU backend" ON)
option(BOVARY_SANITIZE_PATHS "Map build paths in review-only MSVC CPU release binaries" OFF)
set(BOVARY_ENGINE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/work/engine/bovary-llama.cpp" CACHE PATH "Pinned patched llama.cpp source")
if(NOT EXISTS "${BOVARY_ENGINE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Engine missing. Run: python scripts/bootstrap_engine.py")
endif()
if(BOVARY_SANITIZE_PATHS)
    if(NOT BOVARY_CPU_ONLY OR NOT CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
        message(FATAL_ERROR "Path-sanitized release builds currently require the verified MSVC CPU configuration")
    endif()
    file(TO_NATIVE_PATH "${CMAKE_CURRENT_SOURCE_DIR}" BOVARY_NATIVE_SOURCE_DIR)
    add_compile_options("$<$<COMPILE_LANGUAGE:C,CXX>:/experimental:deterministic>"
                        "$<$<COMPILE_LANGUAGE:C,CXX>:/pathmap:${BOVARY_NATIVE_SOURCE_DIR}=.>")
endif()
if(BOVARY_CPU_ONLY)
    set(GGML_CPU ON CACHE BOOL "" FORCE)
    set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
    foreach(backend CUDA HIP VULKAN SYCL OPENCL METAL RPC CANN MUSA WEBGPU
                    HEXAGON VIRTGPU_FRONTEND OPENVINO ZDNN ZENDNN ET BLAS ACCELERATE)
        set(GGML_${backend} OFF CACHE BOOL "" FORCE)
    endforeach()
endif()
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${BOVARY_ENGINE_DIR}" engine)
add_executable(bovary-infer src/infer.cpp src/native_paths.h)
target_link_libraries(bovary-infer PRIVATE llama)
target_compile_features(bovary-infer PRIVATE cxx_std_17)
if(BOVARY_CPU_ONLY)
    target_compile_definitions(bovary-infer PRIVATE BOVARY_CPU_ONLY=1)
endif()
add_executable(bovary-quantize src/quantize.cpp src/native_paths.h)
target_link_libraries(bovary-quantize PRIVATE llama)
target_compile_features(bovary-quantize PRIVATE cxx_std_17)
add_executable(bovary-paths-test EXCLUDE_FROM_ALL tests/fixtures/native_paths.cpp src/native_paths.h)
target_compile_features(bovary-paths-test PRIVATE cxx_std_17)
target_include_directories(bovary-paths-test PRIVATE "${BOVARY_ENGINE_DIR}/src")
target_link_libraries(bovary-paths-test PRIVATE llama)
if(MINGW)
    foreach(target bovary-infer bovary-quantize bovary-paths-test)
        target_link_options(${target} PRIVATE -municode)
    endforeach()
endif()
